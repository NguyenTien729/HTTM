const h = (tag, attrs = {}, ...kids) => {
  const e = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs)) {
    if (k === 'class') e.className = v;
    else if (k.startsWith('on')) e[k] = v;
    else if (v !== false && v != null) e.setAttribute(k, v === true ? '' : v);
  }
  for (const c of kids.flat().filter(c => c !== false && c != null)) e.append(c.nodeType ? c : document.createTextNode(c));
  return e;
};
async function api(method, path, body) {
  const r = await fetch(path, { method, headers: body ? { 'Content-Type': 'application/json' } : {}, body: body ? JSON.stringify(body) : undefined });
  if (!r.ok) {
    let d; try { d = (await r.json()).detail } catch {}
    throw new Error(typeof d === 'string' ? d : (d ? JSON.stringify(d) : 'Lỗi HTTP ' + r.status));
  }
  return r;
}

// ------------------------------------------------------------------ ENGINE
const cards = STEPS.map((st, i) => {
  const dot = h('div', { class: 'dot' }, i + 1), ttl = h('span', { class: 'ttl' }, st.t), tag = st.input ? h('span', { class: 'tag' }, 'cần bạn') : '';
  const sum = h('div', { class: 'sum' }), body = h('div', { class: 'body' });
  const el = h('div', { class: 'step pending' }, dot, h('div', { class: 'card' }, ttl, tag, sum, body));
  $('#steps').append(el);
  return { el, dot, sum, body };
});
function status(i, s, text) {
  const c = cards[i];
  c.el.className = 'step ' + s;
  c.dot.textContent = s === 'done' ? '✓' : s === 'err' ? '!' : s === 'wait' ? '?' : i + 1;
  if (text != null) c.sum.textContent = text;
}
STEPS.forEach((_, i) => { if (i < S.i) status(i, 'done', S.sum[i] || ''); });

let busy = false;
async function loop() {
  if (busy) return;
  busy = true;
  while (S.i < STEPS.length) {
    const i = S.i, st = STEPS[i], card = cards[i];
    status(i, st.input ? 'wait' : 'run', st.input ? 'Cần bạn quyết định để tiếp tục' : 'Đang chạy…');
    card.body.replaceChildren();
    card.el.scrollIntoView({ behavior: 'smooth', block: 'center' });
    try {
      const sum = await st.run(S.ctx, card.body);
      S.sum[i] = sum || '';
      status(i, 'done', S.sum[i]);
      S.i++; save();
    } catch (e) {
      status(i, 'err', 'Lỗi');
      const again = h('button', { class: 'pri', onclick: () => { busy = false; loop() } }, 'Thử lại');
      card.body.append(h('div', { class: 'err' }, e.message), again);
      if (st.back != null) card.body.append(h('button', { onclick: () => { S.i = st.back; save(); STEPS.forEach((_, k) => k >= S.i && status(k, 'pending', '')); busy = false; loop() } }, 'Quay lại bước chọn'));
      busy = false; return;
    }
  }
  busy = false;
  $('#steps').append(h('button', { class: 'pri', onclick: () => $('#reset').click() }, 'Bắt đầu nghiên cứu mới'));
}
$('#reset').onclick = () => { if (confirm('Xoá tiến trình hiện tại và bắt đầu lại?')) { try { localStorage.removeItem(KEY) } catch {} location.reload() } };
loop();
